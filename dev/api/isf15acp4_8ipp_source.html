51template <typename SpiType>
52void SmartDisplay<SpiType>::pulseReset() noexcept {
53 // SSD1331 rev 1.2, section 7.9 and Table 18: hold RES# low (Reset asserted = Active) for at least 3 us, release
54 // it high (Inactive), then allow the reset completion time (2 us max) before the first command.
55 bus_->GpioSetActive(CtrlPin::Reset);
56 bus_->DelayUs(ssd1331::kResetPulseMinUs);
57 bus_->GpioSetInactive(CtrlPin::Reset);
58 bus_->DelayUs(ssd1331::kResetRecoveryUs);
59}

61template <typename SpiType>
62bool SmartDisplay<SpiType>::sendCommand(uint8_t cmd, const uint8_t* data, uint8_t len) noexcept {
63 if (!bus_->BeginCommand()) {
64 setError(Error::SpiWrite);
65 return false;
66 }
67
68 if (!bus_->WriteBytes(&cmd, 1)) {
69 bus_->EndTransaction();
70 setError(Error::SpiWrite);
71 return false;
72 }
73
74 if (len > 0 && data != nullptr) {
75 if (!bus_->WriteBytes(data, len)) {
76 bus_->EndTransaction();
77 setError(Error::SpiWrite);
78 return false;
79 }
80 }
81
82 bus_->EndTransaction();
83 return true;
84}

109template <typename SpiType>
110bool SmartDisplay<SpiType>::Initialize(bool enable_vcc) noexcept {
111 if (bus_ == nullptr) {
112 setError(Error::InvalidParam);
113 return false;
114 }
115
116 if (!bus_->EnsureInitialized()) {
117 setError(Error::NotInitialized);
118 return false;
119 }
120
121 bus_->GpioSetInactive(CtrlPin::ChipSelect);
122 bus_->GpioSetActive(CtrlPin::DataCommand);
123 bus_->GpioSetInactive(CtrlPin::VccEnable);
124
125 pulseReset();
126
127 if (enable_vcc) {
128 bus_->GpioSetActive(CtrlPin::VccEnable);
129 bus_->DelayUs(1000);

134 }
135
136 if (!SetFullWindow()) {
137 return false;
138 }
139
140 initialized_ = true;
141 return true;

174template <typename SpiType>
175bool SmartDisplay<SpiType>::SetWindow(const Rect& rect) noexcept {
176 if (rect.x1 < rect.x0 || rect.y1 < rect.y0 || rect.x1 >= geometry_.width ||
177 rect.y1 >= geometry_.height) {
178 setError(Error::OutOfRange);
179 return false;
180 }
181
182 // Display coordinates to controller RAM addresses: the ISC15ANP4 and ISC01P show a window of the 96 x 64 RAM
183 // starting at column 10h / 16h (NKK Application Notes rev H, Tables 4 and 5).
184 const Rect ram = toRam(rect);
185 const uint8_t col_data[2] = {ram.x0, ram.x1};
186 const uint8_t row_data[2] = {ram.y0, ram.y1};
187
188 if (!sendCommand(static_cast<uint8_t>(ssd1331::Command::SetColumnAddress), col_data, 2)) {
189 return false;
190 }
191 if (!sendCommand(static_cast<uint8_t>(ssd1331::Command::SetRowAddress), row_data, 2)) {
192 return false;
193 }
194
195 window_ = rect;
196 return true;
197}

236template <typename SpiType>
237bool SmartDisplay<SpiType>::WriteColors(const Color565* colors, std::size_t count) noexcept {
238 if (!initialized_) {
239 setError(Error::NotInitialized);
240 return false;
241 }
242 if (colors == nullptr || count == 0) {
243 setError(Error::InvalidParam);
244 return false;
245 }
246
247 if (!bus_->BeginData()) {
248 setError(Error::SpiWrite);
249 return false;
250 }
251
252 // SSD1331 rev 1.2 section 7.5.2, Table 9 (65k colour, 8-bit interface): first byte C4..C0 B5..B3, second byte
253 // B2..B0 A4..A0, i.e. the 16-bit pixel high byte first, with C = blue and A = red on the NKK panels
254 // (ToPanelPixel). Serialise explicitly so the CPU's byte order does not matter, in chunks inside one data
255 // transaction.
256 constexpr std::size_t kChunkPixels = 64;
257 uint8_t chunk[kChunkPixels * 2];
258 std::size_t done = 0;
259 while (done < count) {
260 const std::size_t n = (count - done) < kChunkPixels ? (count - done) : kChunkPixels;
261 for (std::size_t i = 0; i < n; ++i) {
262 const uint16_t c = ssd1331::ToPanelPixel(colors[done + i]);
263 chunk[2 * i] = static_cast<uint8_t>(c >> 8);
264 chunk[2 * i + 1] = static_cast<uint8_t>(c & 0xFF);
265 }
266 if (!bus_->WriteBytes(chunk, n * 2)) {
267 bus_->EndTransaction();
268 setError(Error::SpiWrite);
269 return false;
270 }
271 done += n;
272 }
273
274 bus_->EndTransaction();
275 return true;
276}

339template <typename SpiType>
340void SmartDisplay<SpiType>::colorToSsd1331(Color565 color, uint8_t out[3]) noexcept {
341 // NKK panels: colour A = red, B = green, C = blue (NKK Application Notes rev H section 9; init A0h bit 2 = 0).
342 out[0] = static_cast<uint8_t>((color >> 11) & 0x1F); // A: red 5-bit
343 out[1] = static_cast<uint8_t>((color >> 5) & 0x3F); // B: green 6-bit
344 out[2] = static_cast<uint8_t>(color & 0x1F); // C: blue 5-bit
345}

358template <typename SpiType>
359bool SmartDisplay<SpiType>::DrawRectangle(const Rect& rect, Color565 color, bool filled) noexcept {
360 if (filled) {
361 const uint8_t fill_cfg[2] = {0x01, 0x00};
362 if (!sendCommand(static_cast<uint8_t>(ssd1331::Command::FillEnable), fill_cfg, 2)) {
363 return false;
364 }
365 }
366
367 uint8_t rgb[3];
368 colorToSsd1331(color, rgb);
369 const Rect ram = toRam(rect);
370 const uint8_t data[10] = {ram.x0, ram.y0, ram.x1, ram.y1,
371 static_cast<uint8_t>(rgb[2] << 1), rgb[1],
372 static_cast<uint8_t>(rgb[0] << 1),
373 static_cast<uint8_t>(rgb[2] << 1), rgb[1],
374 static_cast<uint8_t>(rgb[0] << 1)};
375 const bool ok = sendCommand(static_cast<uint8_t>(ssd1331::Command::DrawRectangle), data,
376 sizeof(data));
377
378 if (filled) {
379 const uint8_t disable[2] = {0x00, 0x00};
380 sendCommand(static_cast<uint8_t>(ssd1331::Command::FillEnable), disable, 2);
381 }
382 return ok;

397template <typename SpiType>
398bool SmartDisplay<SpiType>::ConfigureScroll(uint8_t h_offset, uint8_t start_row, uint8_t scroll_rows,
399 uint8_t v_offset, uint8_t interval_code,
400 bool activate) noexcept {
401 // SSD1331 rev 1.2 Table 10, 27h note 2: the parameters must not change while scrolling is active, so stop
402 // first (NKK Application Notes rev H section 12: send 27h only with scrolling deactivated).
403 if (!StopScroll()) {
404 return false;
405 }
406 const uint8_t data[5] = {h_offset, start_row, scroll_rows, v_offset,
407 static_cast<uint8_t>(interval_code & 0x03)};
408 if (!sendCommand(static_cast<uint8_t>(ssd1331::Command::ScrollSetup), data, sizeof(data))) {
409 return false;
410 }
411 if (!activate) {
412 return true;
413 }
414 return sendCommandOnly(static_cast<uint8_t>(ssd1331::Command::ActivateScroll));
415}
